# He3 ADR Cycle Custom Script
# 0. Close green valve check
print("CLOSE THE GREEN HE3 VALVE")

# 1. Close all heatswitches in the beginning and wait 30 seconds
close heatswitch pot
wait 1 s
close heatswitch adr
wait 1 s
close heatswitch charcoal
wait 30 s

# Pre-cooling check: wait until 3K starting temp < 3.2K (adaptive slow polling)
while True:
    mr = most_recent_measurements()
    c_temp = mr.get("cryocon_chC_temperature", 100)
    if c_temp < 3.2:
        break
    world.wait(60 if c_temp > 20 else (30 if c_temp > 5 else 10))

# 2. Switch relay to RAMP and ensure current starts from zero
set relay RAMP
wait 1 s
ramp_controller = world.station.ls370.heater
ramp_controller.mode('open_loop')
ramp_controller.out(0.0)
ramp_controller.range("10mA")
wait 1 s

# 3. Open charcoal heatswitch and start heating charcoal
open heatswitch charcoal
wait 1 s
world.station.cryocon.loop1_source("A")
world.wait(1)
world.station.cryocon.loop1_setpoint(65)
world.wait(1)
world.station.cryocon.loop2_source("B")
world.wait(1)
world.station.cryocon.loop2_range("HI")
world.wait(1)
world.station.cryocon.loop2_setpoint(55)
world.wait(1)
world.station.cryocon.control_enabled(True)
world.wait(1)

# 4. Ramp up ADR magnet from zero

target_hout = 58.1
target_time_s = 2700 # 45 min
t_start_ramp = time.time()
while True:
    elapsed = time.time() - t_start_ramp
    fraction = min(1.0, elapsed / target_time_s)
    ramp_controller.out(fraction * target_hout)
    if fraction >= 1.0:
        break
    world.wait(1)

# 4. Wait for He3 to condense
world.wait(12600)

# 5. Cool charcoal to launch temperature (< 1.0 K)
open heatswitch pot
world.wait(1)
world.station.cryocon.control_enabled(False)
world.wait(120)
close heatswitch charcoal
world.wait(1)

while True:
    mr = most_recent_measurements()
    if mr.get("faa_temperature", 100) < 1.0:
        break
    world.wait(1)

# 6. Ramp down ADR
open heatswitch adr
world.wait(1)
world.wait(600) # 10 min launch settle after opening ADR switch before ramp down
t_start_down = time.time()
target_down_time_s = 2700 # 45 min
while True:
    elapsed = time.time() - t_start_down
    fraction = min(1.0, elapsed / target_down_time_s)
    ramp_controller.out((1.0 - fraction) * target_hout)
    if fraction >= 1.0:
        break
    world.wait(1)

world.wait(1200)
set relay CONTROL
world.wait(30)
ramp_controller.mode('closed')
world.wait(1)
ramp_controller.setpoint(0.05)
world.wait(1)
ramp_controller.range("10mA")

return wait_forever
